Dual-track reads

When “success” is the bug.

After a migration a PostgreSQL table can be empty while the JSON store still holds production data. A query that “succeeds” with zero rows, if trusted, makes the whole admin console look wiped. The dual-track layer decides — per domain — when PostgreSQL is authoritative and when to fall back, so the rollout could be staged, and reversed, with one environment variable.

server/services/dual-track.js
// PG has rows             → use PG
// PG table missing        → fall back to JSON, or throw if authoritative
// PG empty, JSON has data → fall back, or trust PG if authoritative
//                           (a deleted row can never be "resurrected")

function isPgAuthoritative(domain) {
  if (isTruthyEnv('PG_JSON_FALLBACK')) return false; // escape hatch
  if (isTruthyEnv('PG_AUTHORITATIVE')) return true;
  if (process.env.NODE_ENV === 'production') return true;
  const list = String(process.env.PG_AUTHORITATIVE_DOMAINS || '')
    .split(',').map((s) => s.trim()).filter(Boolean);
  return !!(domain && list.includes(domain));
}

async function loadListPreferPg({ loadPg, loadJson, domain, autoSeed }) {
  const authoritative = isPgAuthoritative(domain);
  let pgList;
  try { pgList = await loadPg(); }
  catch (err) {
    // only a missing table may fall back — never a network or auth error
    if (!isMissingRelation(err) || authoritative) throw err;
    return normalizeList(loadJson());
  }
  if (pgList.length > 0) return pgList;
  const jsonList = normalizeList(loadJson());
  if (jsonList.length && autoSeed) await autoSeed(jsonList); // idempotent
  return authoritative ? pgList : jsonList;
}
Orphan payments

Callbacks don't respect order state.

A customer cancels; seconds later the gateway's delayed notification lands and the money is now attached to an order that no longer exists. Instead of a manual reconciliation, the callback path recognises any payment that arrives against a non-pending order and refunds the surplus through the original channel — under a row lock, with a ledger entry, and without ever throwing, so the gateway always receives its acknowledgement.

server/services/orphan-refund.js · condensed
// Runs inside the payment callback. Never throws: a failed refund is
// logged and retried, so the gateway always gets its ACK.

const SURPLUS_REFUNDABLE = new Set([
  'cancelled', 'pending_ship', 'shipped', 'pending_receive',
  'completed', 'refunded', 'refunding', 'offline_pending_review'
]);

async function refundOrphanPayment({ Pg, appendLog }, orderId, paymentId) {
  const payment = await loadPayment(Pg, paymentId);
  if (payment?.order_id !== orderId) return skip();
  if (payment.status !== 'paid') return skip();

  const order = await loadOrder(Pg, orderId);
  // a pending order means this is the first, legitimate payment
  if (!SURPLUS_REFUNDABLE.has(order.status)) return skip();

  return Pg.withTx(async (client) => {
    const { rows: [row] } = await client.query(
      `SELECT amount, status FROM order_payments
        WHERE id = $1 FOR UPDATE`, [payment.id]);
    const used = await sumRefundedOrPending(client, payment.id);
    const remaining = row.amount - used;
    if (remaining <= 0.001) return skip(); // idempotent on replay

    await markRefunded(client, payment.id);
    await writeLedger(client, order, remaining, 'orphan_payment');
    return refundViaOriginalChannel(payment, remaining); // wechat | balance
  }).catch((err) => appendLog('orphan-refund-failed', { orderId, err }));
}
